47void WebSocketClientImpl::connectInternal() {
48 // WebSocket 클라이언트 설정
49 websocket_client_config config;
50
51 // WSS 연결을 위한 SSL 설정
52 if (m_uri.find("wss://") == 0) {
53 config.set_validate_certificates(false); // 개발용 - 실제 환경에서는 true로 설정
54 }
55
56 // 새로운 클라이언트 생성
57 m_client = std::make_shared<websocket_callback_client>(config);
58
59 // 이벤트 핸들러 설정
60 setupEventHandlers();
61
62 try {
63 // 연결 시도
64 const auto res = m_client->connect(m_uri)
65 .then([this] {
66 m_connected = true;
67 MOSAIC_LOG_INFO("WebSocket connected to {}", m_uri);
68
69 if (m_onConnected) {
70 m_onConnected();
71 }
72 })
73 .wait();
74 if (res != pplx::task_status::completed) {
75 throw std::runtime_error("Failed to connect to WebSocket server");
76 }
77 } catch (const std::exception& e) {
78 MOSAIC_LOG_ERROR("Connection failed: {}", e.what());
79 }
80}

139void WebSocketClientImpl::handleMessage(const websocket_incoming_message& message) const {
140 try {
141 if (message.message_type() == websocket_message_type::ping) {
142 MOSAIC_LOG_DEBUG("Received ping message");
143 auto pong_message = websocket_outgoing_message();
144 pong_message.set_pong_message("pong");
145 m_client->send(pong_message).wait();
146 return; // Ping 메시지는 Pong으로 응답하고 무시
147 }
148 if (message.message_type() == websocket_message_type::pong) {
149 MOSAIC_LOG_DEBUG("Received pong message");
150 return; // Pong 메시지는 무시
151 }
152 if (message.message_type() == websocket_message_type::close) {
153 m_client->close().wait();
154 MOSAIC_LOG_DEBUG("Received close message from server");
155 return;
156 }
157 if (message.message_type() == websocket_message_type::text_message) {
158 auto ts = message.extract_string()
159 .then([this](const std::string& body) {
160 if (m_onMessage) {
161 m_onMessage(body);
162 }
163 })
164 .wait();
165 if (ts != pplx::task_status::completed) {
166 m_onError("Task did not complete successfully while handling message.");
167 }
168 }
169 } catch (const std::exception& e) {
170 const std::string error = "Message handling error: " + std::string(e.what());
171 if (m_onError) {
172 m_onError(error);
173 }
174 }
175}

177void WebSocketClientImpl::handleClose(websocket_close_status close_status,
178 const std::string& reason,
179 const std::error_code&) {
180 m_connected = false;
181
182 /*
183 *enum class websocket_close_status
184 {
185 normal = 1000,
186 going_away = 1001,
187 protocol_error = 1002,
188 unsupported = 1003, // or data_mismatch
189 abnormal_close = 1006,
190 inconsistent_datatype = 1007,
191 policy_violation = 1008,
192 too_large = 1009,
193 negotiate_error = 1010,
194 server_terminate = 1011,
195 };
196 */
197 MOSAIC_LOG_INFO("WebSocket closed. Status: {} Reason: {} Error Code: {}",
198 static_cast<int>(close_status),
199 reason,
200 std::error_code().message());
201
202 if (m_onDisconnected) {
203 m_onDisconnected();
204 }
205
206 // Reconnection logic start
207 std::thread reconnect_thread([this] {
208 MOSAIC_LOG_ERROR("Reconnecting to WebSocket server in 5 seconds...");
209
210 std::this_thread::sleep_for(std::chrono::seconds(5)); // Wait before reconnecting
211 this->connectInternal();
212 });
213 reconnect_thread.detach();
214}
