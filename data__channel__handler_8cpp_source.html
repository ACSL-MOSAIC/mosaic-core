64 void RegisterDataChannelObserver(IDataChannelHandler* dc_handler) {
65 if (!dc_interface_) {
66 MOSAIC_LOG_ERROR("DataChannelInterface cannot be set for label: {}", label_);
67 throw std::runtime_error("DataChannelInterface is not set");
68 }
69
70 std::function<void(const webrtc::DataBuffer&)> onMessageLambda;
71
72 if (auto receivable_ptr = dynamic_cast<IDataChannelReceivable*>(dc_handler)) {
73 MOSAIC_LOG_DEBUG("Receivable DataChannel Detected!");
74 onMessageLambda = [receivable_ptr](const webrtc::DataBuffer& buffer) { receivable_ptr->OnMessage(buffer); };
75 } else {
76 onMessageLambda = [](const webrtc::DataBuffer&) {};
77 }
78
79 observer_ = std::make_shared<DataChannelObserver>(onMessageLambda);
80 dc_interface_->RegisterObserver(observer_.get());
81 }
82
83 ChannelState GetState() const {
84 if (!dc_interface_) {
85 return kUnknown;
86 }
87
88 switch (dc_interface_->state()) {
89 case webrtc::DataChannelInterface::kConnecting:
90 return kConnecting;
91 case webrtc::DataChannelInterface::kOpen:
92 return kOpen;
93 case webrtc::DataChannelInterface::kClosing:
94 return kClosing;
95 case webrtc::DataChannelInterface::kClosed:
96 return kClosed;
97 default:
98 return kUnknown;
99 }
100 }

123 void SendAsync(const webrtc::DataBuffer& buffer) const {
124 std::lock_guard<std::mutex> lock(queue_mutex_);
125 send_queue_.push(buffer);
126 ProcessQueue();
127 }
128
129 void ProcessQueue() const {
130 if (is_sending_ || send_queue_.empty()) {
131 return;
132 }
133
134 is_sending_ = true;
135 auto buffer = send_queue_.front();
136 send_queue_.pop();
137
138 MOSAIC_LOG_DEBUG("Buffer Size: {}", buffer.size());
139
140 dc_interface_->SendAsync(buffer, [this](webrtc::RTCError error) {
141 std::lock_guard<std::mutex> lock(queue_mutex_);
142 is_sending_ = false;
143
144 if (!error.ok()) {
145 MOSAIC_LOG_ERROR("SendAsync failed for label: {}, error: {}", label_, error.message());
146 }
147
148 ProcessQueue();
149 });
150 }
