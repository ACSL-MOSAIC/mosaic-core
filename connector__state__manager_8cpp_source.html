27void ConnectorStateManager::StateBroadcastWorker() {
28 while (!stop_flag_) {
29 std::unique_lock<std::shared_mutex> lock(mutex_);
30 cv_.wait_for(lock, std::chrono::seconds(1), [this] { return stop_flag_.load(); });

32 SendState();
